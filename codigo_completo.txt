//=================================================
// INICIALIZAÇÃO
//=================================================

void iniciarAutomacao()
{
    pinMode(RELE_LUZ, OUTPUT);

    pinMode(RELE_VENTILADOR, OUTPUT);

    pinMode(RELE_BOMBA, OUTPUT);


    digitalWrite(RELE_LUZ, LOW);

    digitalWrite(RELE_VENTILADOR, LOW);

    digitalWrite(RELE_BOMBA, LOW);


    atuadores.luz = false;

    atuadores.ventilador = false;

    atuadores.aeracao = false;


    Serial.println("Automacao inicializada.");
}


//=================================================
// ILUMINAÇÃO
//=================================================

void controlarIluminacao()
{
    bool ligar = luzLigada();

    atuadores.luz = ligar;

    digitalWrite(
        RELE_LUZ,
        ligar ? HIGH : LOW
    );
}


//=================================================
// VENTILAÇÃO
//=================================================

void controlarVentilacao()
{
    bool ligar = false;


    if(
        ambiente.temperaturaAr >
        cultivoAtual.temperaturaMax
    )
    {
        ligar = true;
    }


    if(
        ambiente.umidadeAr >
        cultivoAtual.umidadeMax
    )
    {
        ligar = true;
    }


    atuadores.ventilador = ligar;


    digitalWrite(
        RELE_VENTILADOR,
        ligar ? HIGH : LOW
    );
}


//=================================================
// AERAÇÃO DWC
//=================================================

void controlarAeracao()
{
    /*
        Na V1 a aeração representa:

        bomba de ar
             |
          mangueira
             |
        pedra porosa
             |
        solução nutritiva

        O controle inteligente da aeração
        será desenvolvido posteriormente.

        Por enquanto ela permanece ligada
        quando o sistema está operacional.
    */


    bool ligar = solucao.nivelOK;


    atuadores.aeracao = ligar;


    digitalWrite(
        RELE_BOMBA,
        ligar ? HIGH : LOW
    );
}


//=================================================
// AUTOMAÇÃO COMPLETA
//=================================================

void atualizarAutomacao()
{
    controlarIluminacao();

    controlarVentilacao();

    controlarAeracao();
}

//=================================================
// INICIALIZAÇÃO
//=================================================

void iniciarDiagnostico()
{
    Serial.println("Diagnostico inicializado.");

    limparAlertas();
}


//=================================================
// LIMPAR ALERTAS
//=================================================

void limparAlertas()
{
    alertas.phBaixo = false;
    alertas.phAlto = false;

    alertas.ecBaixo = false;
    alertas.ecAlto = false;

    alertas.temperaturaBaixa = false;
    alertas.temperaturaAlta = false;

    alertas.umidadeBaixa = false;
    alertas.umidadeAlta = false;

    alertas.nivelBaixo = false;
}


//=================================================
// pH
//=================================================

void diagnosticarPH()
{
    if(solucao.ph < cultivoAtual.phMin)
        alertas.phBaixo = true;

    if(solucao.ph > cultivoAtual.phMax)
        alertas.phAlto = true;
}


//=================================================
// EC
//=================================================

void diagnosticarEC()
{
    if(solucao.ec < cultivoAtual.ecMin)
        alertas.ecBaixo = true;

    if(solucao.ec > cultivoAtual.ecMax)
        alertas.ecAlto = true;
}


//=================================================
// TEMPERATURA
//=================================================

void diagnosticarTemperatura()
{
    if(
        ambiente.temperaturaAr <
        cultivoAtual.temperaturaMin
    )
    {
        alertas.temperaturaBaixa = true;
    }

    if(
        ambiente.temperaturaAr >
        cultivoAtual.temperaturaMax
    )
    {
        alertas.temperaturaAlta = true;
    }
}


//=================================================
// UMIDADE
//=================================================

void diagnosticarUmidade()
{
    if(
        ambiente.umidadeAr <
        cultivoAtual.umidadeMin
    )
    {
        alertas.umidadeBaixa = true;
    }

    if(
        ambiente.umidadeAr >
        cultivoAtual.umidadeMax
    )
    {
        alertas.umidadeAlta = true;
    }
}


//=================================================
// NÍVEL
//=================================================

void diagnosticarNivel()
{
    if(!solucao.nivelOK)
        alertas.nivelBaixo = true;
}


//=================================================
// SAÚDE
//=================================================

void calcularSaude()
{
    int pontos = 100;


    if(
        alertas.phBaixo ||
        alertas.phAlto
    )
    {
        pontos -= 20;
    }


    if(
        alertas.ecBaixo ||
        alertas.ecAlto
    )
    {
        pontos -= 20;
    }


    if(
        alertas.temperaturaBaixa ||
        alertas.temperaturaAlta
    )
    {
        pontos -= 20;
    }


    if(
        alertas.umidadeBaixa ||
        alertas.umidadeAlta
    )
    {
        pontos -= 15;
    }


    if(alertas.nivelBaixo)
    {
        pontos -= 25;
    }


    if(pontos < 0)
        pontos = 0;


    saudeCultivo = pontos;
}


//=================================================
// DIAGNÓSTICO COMPLETO
//=================================================

void atualizarDiagnostico()
{
    limparAlertas();

    diagnosticarPH();

    diagnosticarEC();

    diagnosticarTemperatura();

    diagnosticarUmidade();

    diagnosticarNivel();

    calcularSaude();
}

void iniciarDisplay()
{
    Wire.begin(
        OLED_SDA,
        OLED_SCL
    );


    if(
        !tela.begin(
            SSD1306_SWITCHCAPVCC,
            OLED_ENDERECO
        )
    )
    {
        Serial.println(
            "ERRO: OLED nao encontrado."
        );

        return;
    }


    tela.clearDisplay();

    tela.setTextColor(
        SSD1306_WHITE
    );

    tela.display();


    Serial.println(
        "Display inicializado."
    );
}

//=====================================================
// CONTROLE DE TEMPO
//=====================================================

unsigned long ultimoCiclo = 0;


//=====================================================
// SETUP
//=====================================================

void setup()
{
    Serial.begin(115200);

    delay(500);


    Serial.println();

    Serial.println("================================");
    Serial.println("        HORTA DO FUTURO");
    Serial.println("      COMPUTADOR DE PLANTAS");
    Serial.println("          Firmware V1.0");
    Serial.println("================================");


    //=============================================
    // SISTEMA
    //=============================================

    iniciarSistema();


    //=============================================
    // OLED
    //=============================================

    iniciarDisplay();


    //=============================================
    // TELA INICIAL
    //=============================================

    mostrarInicializacao();


    delay(1500);


    //=============================================
    // MONITORAMENTO
    //=============================================

    estadoAtual = MONITORAMENTO;
}


//=====================================================
// LOOP
//=====================================================

void loop()
{
    unsigned long agora = millis();


    //=============================================
    // CICLO DE MONITORAMENTO
    //=============================================

    if(
        agora - ultimoCiclo >=
        INTERVALO_CICLO
    )
    {
        ultimoCiclo = agora;

        executarCiclo();
    }


    //=============================================
    // INTERFACE
    //=============================================

    atualizarInterface();


    //=============================================
    // MENU
    //=============================================

    atualizarMenu();
}

#ifndef MENU_H
#define MENU_H

//=================================================
// MENU
//=================================================

int opcaoMenu = 0;

const int totalOpcoes = 5;


String opcoesMenu[] =
{
    "Monitoramento",
    "Cultivos",
    "Configuracoes",
    "Historico",
    "Sistema"
};


//=================================================
// MOSTRA MENU
//=================================================

void mostrarMenu()
{
    tela.clearDisplay();

    tela.setTextSize(1);

    tela.setCursor(0, 0);

    tela.println("MENU PRINCIPAL");


    for(
        int i = 0;
        i < totalOpcoes;
        i++
    )
    {
        tela.setCursor(
            0,
            14 + (i * 10)
        );


        if(i == opcaoMenu)
            tela.print("> ");
        else
            tela.print("  ");


        tela.println(
            opcoesMenu[i]
        );
    }


    tela.display();
}


//=================================================
// CIMA
//=================================================

void menuCima()
{
    opcaoMenu--;


    if(opcaoMenu < 0)
        opcaoMenu =
            totalOpcoes - 1;
}


//=================================================
// BAIXO
//=================================================

void menuBaixo()
{
    opcaoMenu++;


    if(
        opcaoMenu >= totalOpcoes
    )
    {
        opcaoMenu = 0;
    }
}


//=================================================
// SELECIONAR
//=================================================

void selecionarMenu()
{
    switch(opcaoMenu)
    {
        case 0:
            estadoAtual =
                MONITORAMENTO;
            break;

        case 1:
            estadoAtual =
                MENU_CULTIVOS;
            break;

        case 2:
            estadoAtual =
                MENU_CONFIGURACOES;
            break;

        case 3:
            estadoAtual =
                MENU_HISTORICO;
            break;

        case 4:
            estadoAtual =
                MENU_SISTEMA;
            break;
    }
}


//=================================================
// VOLTAR
//=================================================

void voltarMenu()
{
    estadoAtual =
        MENU_PRINCIPAL;
}


//=================================================

//=================================================

void atualizarMenu()
{
    /*

//=================================================
// FOTOPERÍODO
//=================================================

bool luzLigada()
{
    int horaLiga = HORA_INICIO_LUZ;

    int horaDesliga =
        horaLiga + cultivoAtual.horasLuz;

    if(horaDesliga < 24)
    {
        return
        (
            horaAtual >= horaLiga &&
            horaAtual < horaDesliga
        );
    }

    return
    (
        horaAtual >= horaLiga ||
        horaAtual < (horaDesliga - 24)
    );
}

/*
=================================================

ÍNDICE DE SAÚDE

Cálculo:

Temperatura 30%
pH           30%
EC           20%
Nível água   20%

=================================================
*/



float avaliarTemperatura()
{


if(
ambiente.temperaturaAr >= cultivoAtual.temperaturaMin
&&
ambiente.temperaturaAr <= cultivoAtual.temperaturaMax
)

{

return 100;

}

else

{

return 50;

}



}



float avaliarPH()
{


if(
ambiente.ph >= cultivoAtual.phMin
&&
ambiente.ph <= cultivoAtual.phMax
)

{

return 100;

}

else

{

return 50;

}


}



float avaliarNivel()
{


if(ambiente.nivelAgua)

return 100;


else

return 0;


}




void calcularSaude()
{


float resultado;



resultado =
(
avaliarTemperatura()*0.30
+
avaliarPH()*0.30
+
90*0.20
+
avaliarNivel()*0.20
);



indiceSaude =
(int)resultado;



}

#ifndef SENSOR_TEMPERATURA_H
#define SENSOR_TEMPERATURA_H



#include "config.h"

#include "dados.h"



#include <DHT.h>

#include <OneWire.h>

#include <DallasTemperature.h>





// ==============================
// DHT22
// ==============================


#define TIPO_DHT DHT22


DHT dht(
PINO_DHT,
TIPO_DHT
);




// ==============================
// DS18B20
// ==============================


OneWire barramentoTemperatura(
PINO_DS18B20
);


DallasTemperature sensorAgua(
&barramentoTemperatura
);





void iniciarSensoresTemperatura()
{


dht.begin();


sensorAgua.begin();


Serial.println(
"Sensores temperatura OK"
);



}






void lerTemperatura()
{


/*

Leitura do ar

*/

float temperatura =
dht.readTemperature();



float umidade =
dht.readHumidity();





if(
!isnan(temperatura)
)

{

sensores.temperaturaAr =
temperatura;

}



if(
!isnan(umidade)
)

{

sensores.umidadeAr =
umidade;

}





/*

Leitura água

*/

sensorAgua.requestTemperatures();



sensores.temperaturaAgua =
sensorAgua.getTempCByIndex(0);




Serial.print(
"Temperatura ar: "
);


Serial.println(
sensores.temperaturaAr
);



}
